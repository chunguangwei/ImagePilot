### 2.2 Service层平台兼容性分析

#### ✅ 完全兼容的服务（无需修改）

1. **ConfigService.js**
   - 纯配置读取逻辑
   - 使用fetch API（React Native支持）
   - **结论**: 无需修改

2. **UnifiedDataService.js**
   - 统一数据访问接口
   - 已通过适配器抽象底层存储
   - **结论**: 无需修改

3. **ColorHistogramExtractor.js**
   - 纯算法实现
   - **结论**: 无需修改

#### ⚠️ 需要适配的服务

1. **ImageClassifierService.js** ❗❗❗
   ```javascript
   // 问题：依赖onnxruntime-node
   import * as ort from 'onnxruntime-node';
   
   // 移动端不支持ONNX Runtime本地推理
   // 需要改为纯远程推理方案
   ```
   
   **问题点**:
   - 使用`onnxruntime-node`（仅Node.js环境）
   - 移动端无法加载本地ONNX模型
   - 依赖Sharp库（Node.js图像处理）
   
   **解决方案**:
   - 移动端完全使用远程推理API
   - 移除本地模型加载逻辑
   - 使用React Native Image库替代Sharp

2. **ImageSimilarityService.js**
   ```javascript
   // 问题：依赖Canvas和Sharp
   const { createCanvas } = require('canvas');
   const sharp = require('sharp');
   ```
   
   **问题点**:
   - 使用Node.js Canvas库
   - 依赖Sharp进行图像处理
   
   **解决方案**:
   - 使用react-native-image-resizer
   - 使用react-native-fs读取图片数据
   - 在JavaScript层实现哈希算法（无需Canvas）

3. **ImageStorageService.js**
   ```javascript
   // 已有适配器模式，但需要优化
   class IndexedDBAdapter { ... }  // Web端
   class AsyncStorageAdapter { ... } // 移动端
   ```
   
   **问题点**:
   - AsyncStorage性能较差（序列化问题）
   - 大量图片数据存储慢
   
   **建议优化**:
   - 移动端改用SQLite（react-native-sqlite-storage）
   - 创建SQLiteAdapter
   - 保持接口兼容

4. **GalleryScannerService.js**
   ```javascript

   const fs = require('fs');
   const path = require('path');
   ```
   
   **问题点**:
   - 使用Node.js fs模块
   - 依赖Electron的文件访问
   - PC端扫描Windows文件系统
   
   **解决方案**:
   - 移动端使用react-native-fs
   - Android使用MediaStore API（已有MediaStoreModule）
   - iOS使用Photos Framework
   - 通过适配器统一接口

5. **CityLocationService.js**
   ```javascript

   ```
   
   **问题点**:
   - 需要网络请求（高德/百度地图API）

```javascript
class SQLiteAdapter {
  constructor() {
    this.db = null;
  }

  async init() {
    const SQLite = require('react-native-sqlite-storage');
    this.db = await SQLite.openDatabase({
      name: 'ImageClassifier.db',
      location: 'default'
    });
    
    // 创建表
    await this.db.executeSql(`
      CREATE TABLE IF NOT EXISTS images (
        id TEXT PRIMARY KEY,
        fileName TEXT,
        uri TEXT,
        category TEXT,
        createdAt INTEGER,
        -- 其他字段...
      )
    `);
    
    await this.db.executeSql(`
      CREATE INDEX IF NOT EXISTS idx_category ON images(category)
    `);
  }

  async getItem(key) {
    if (key === 'images') {
      const [results] = await this.db.executeSql(
        'SELECT * FROM images ORDER BY createdAt DESC'
      );
      return Array.from({ length: results.rows.length }, (_, i) => 
        results.rows.item(i)
      );
    }
    // 其他key的处理...
  }

  async setItem(key, value) {
    if (key === 'images') {
      // 批量插入优化
      const insertSql = `
        INSERT OR REPLACE INTO images 
        (id, fileName, uri, category, createdAt, ...) 
        VALUES (?, ?, ?, ?, ?, ...)
      `;
      
      for (const image of value) {
        await this.db.executeSql(insertSql, [
          image.id, image.fileName, image.uri, 
          image.category, image.createdAt, ...
        ]);
      }
    }
  }
}
```

#### 示例2: ImageClassifierService - 纯远程推理

```javascript
// 移动端版本 - 移除本地模型
class ImageClassifierService {
  constructor() {
    this.remoteEndpoint = 'https://your-server.example.com/classify';
    // 移除 this.ort, this.models 等本地推理相关
  }

  async initialize() {
    // 不需要加载本地模型
    // 只需要检查网络连接
    const isOnline = await this.checkOnlineStatus();
    if (!isOnline) {
      throw new Error('需要网络连接进行图片分类');
    }
  }

  async classifyImage(imageUri) {
    // 直接使用远程推理
    return await this.classifyImageRemote(imageUri);
  }

  async classifyImageRemote(imageUri) {
    // 读取图片文件
    const base64Image = await RNFS.readFile(imageUri, 'base64');
    
    // 调用远程API
    const response = await fetch(this.remoteEndpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        image: base64Image,
        clientId: 'mobile-app'
      })
    });
    
    const result = await response.json();
    return result.category;
  }
}
```

```javascript
// 移动端适配
class GalleryScannerService {
  async scanGallery() {
    // 检测平台
    if (Platform.OS === 'android') {
      return await this.scanAndroidMediaStore();
    } else if (Platform.OS === 'ios') {
      return await this.scanIOSPhotos();
    }
  }

  async scanAndroidMediaStore() {
    // 使用原生MediaStore模块
    const MediaStore = NativeModules.MediaStoreModule;
    
    // 获取所有图片
    const images = await MediaStore.getAllImages();
    
    return images.map(image => ({
      id: image.id,
      fileName: image.fileName,
      uri: image.uri,
      size: image.size,
      mimeType: image.mimeType,
      dateAdded: image.dateAdded,
      dateTaken: image.dateTaken,
      // GPS信息
      latitude: image.latitude,
      longitude: image.longitude
    }));
  }

  async scanIOSPhotos() {
    // iOS使用@react-native-camera-roll/camera-roll
    const CameraRoll = require('@react-native-camera-roll/camera-roll');
    const photos = await CameraRoll.getPhotos({ first: 10000 });
    
    return photos.edges.map(edge => ({
      id: edge.node.image.uri,
      fileName: edge.node.image.filename,
      uri: edge.node.image.uri,
      // ...
    }));
  }
}
```

#### 4.3.2 SQLite适配器实现

```javascript
// src/services/storage/SQLiteAdapter.js
import SQLite from 'react-native-sqlite-storage';

SQLite.enablePromise(true);

class SQLiteAdapter {
  constructor() {
    this.db = null;
    this.isInitialized = false;
  }

  async init() {
    if (this.isInitialized) return this.db;

    try {
      this.db = await SQLite.openDatabase({
        name: 'ImageClassifier.db',
        location: 'default'
      });

      // 创建表
      await this.createTables();
      
      // 数据库优化
      await this.db.executeSql('PRAGMA journal_mode = WAL;'); // 写前日志
      await this.db.executeSql('PRAGMA synchronous = NORMAL;');
      await this.db.executeSql('PRAGMA cache_size = 10000;');

      this.isInitialized = true;
      return this.db;
    } catch (error) {
      console.error('SQLite初始化失败:', error);
      throw error;
    }
  }

  async createTables() {
    const sql = `
      CREATE TABLE IF NOT EXISTS images (
        id TEXT PRIMARY KEY,
        fileName TEXT NOT NULL,
        uri TEXT NOT NULL,
        category TEXT,
        city TEXT,
        size INTEGER,
        mimeType TEXT,
        width INTEGER,
        height INTEGER,
        createdAt INTEGER,
        modifiedAt INTEGER,
        takenAt INTEGER,
        latitude REAL,
        longitude REAL,
        isScreenshot INTEGER DEFAULT 0,
        isIdCard INTEGER DEFAULT 0,
        isStaged INTEGER DEFAULT 0,
        isDeleted INTEGER DEFAULT 0
      );

      CREATE INDEX IF NOT EXISTS idx_category ON images(category);
      CREATE INDEX IF NOT EXISTS idx_city ON images(city);
      CREATE INDEX IF NOT EXISTS idx_createdAt ON images(createdAt DESC);
      CREATE INDEX IF NOT EXISTS idx_takenAt ON images(takenAt DESC);
      CREATE INDEX IF NOT EXISTS idx_isStaged ON images(isStaged);
      CREATE INDEX IF NOT EXISTS idx_isDeleted ON images(isDeleted);

      CREATE TABLE IF NOT EXISTS category_stats (
        category TEXT PRIMARY KEY,
        count INTEGER DEFAULT 0,
        lastUpdated INTEGER
      );

      CREATE TABLE IF NOT EXISTS city_stats (
        city TEXT PRIMARY KEY,
        count INTEGER DEFAULT 0,
        lastUpdated INTEGER
      );

      CREATE TABLE IF NOT EXISTS settings (
        key TEXT PRIMARY KEY,
        value TEXT
      );

      CREATE TABLE IF NOT EXISTS classification_cache (
        imageHash TEXT PRIMARY KEY,
        category TEXT,
        confidence REAL,
        createdAt INTEGER,
        expiresAt INTEGER
      );
    `;

    const statements = sql.split(';').filter(s => s.trim());
    for (const statement of statements) {
      await this.db.executeSql(statement);
    }
  }

  // 批量插入图片（使用事务）
  async batchInsertImages(images) {
    await this.init();
    
    try {
      await this.db.transaction(async (tx) => {
        const sql = `
          INSERT OR REPLACE INTO images (
            id, fileName, uri, category, city, size, mimeType,
            width, height, createdAt, modifiedAt, takenAt,
            latitude, longitude, isScreenshot, isIdCard, isStaged, isDeleted
          ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        `;

        for (const image of images) {
          await tx.executeSql(sql, [
            image.id,
            image.fileName,
            image.uri,
            image.category,
            image.city,
            image.size,
            image.mimeType,
            image.width,
            image.height,
            image.createdAt,
            image.modifiedAt,
            image.takenAt,
            image.latitude,
            image.longitude,
            image.isScreenshot ? 1 : 0,
            image.isIdCard ? 1 : 0,
            image.isStaged ? 1 : 0,
            image.isDeleted ? 1 : 0
          ]);
        }
      });

      // 更新统计
      await this.updateCategoryStats();
      await this.updateCityStats();
    } catch (error) {
      console.error('批量插入失败:', error);
      throw error;
    }
  }

  // 按分类查询图片（带分页）
  async getImagesByCategory(category, offset = 0, limit = 50) {
    await this.init();

    const sql = `
      SELECT * FROM images 
      WHERE category = ? AND isDeleted = 0
      ORDER BY takenAt DESC, createdAt DESC
      LIMIT ? OFFSET ?
    `;

    const [result] = await this.db.executeSql(sql, [category, limit, offset]);
    
    const images = [];
    for (let i = 0; i < result.rows.length; i++) {
      images.push(result.rows.item(i));
    }

    return images;
  }

  // 获取分类统计
  async getCategoryStats() {
    await this.init();

    const sql = `
      SELECT category, COUNT(*) as count 
      FROM images 
      WHERE isDeleted = 0 
      GROUP BY category
    `;

    const [result] = await this.db.executeSql(sql);
    
    const stats = {};
    for (let i = 0; i < result.rows.length; i++) {
      const row = result.rows.item(i);
      stats[row.category] = row.count;
    }

    return stats;
  }

  // 更新分类统计（使用触发器更快，但这里手动更新）
  async updateCategoryStats() {
    await this.init();

    const sql = `
      INSERT OR REPLACE INTO category_stats (category, count, lastUpdated)
      SELECT category, COUNT(*), ?
      FROM images
      WHERE isDeleted = 0
      GROUP BY category
    `;

    await this.db.executeSql(sql, [Date.now()]);
  }

  // 其他方法...
}

export default new SQLiteAdapter();
```

#### 4.3.3 缓存层设计

```javascript
// src/services/storage/CacheManager.js
class CacheManager {
  constructor() {
    this.memoryCache = new Map(); // 内存缓存
    this.maxMemoryCacheSize = 500; // 最多缓存500条记录
  }

  // 获取数据（内存缓存 → SQLite）
  async get(key) {
    // 1. 先查内存缓存
    if (this.memoryCache.has(key)) {
      return this.memoryCache.get(key);
    }

    // 2. 查SQLite
    const value = await SQLiteAdapter.getSetting(key);
    
    // 3. 写入内存缓存
    if (value) {
      this.setMemoryCache(key, value);
    }

    return value;
  }

  // 设置数据（同步写入内存和SQLite）
  async set(key, value) {
    // 1. 写入内存缓存
    this.setMemoryCache(key, value);

    // 2. 写入SQLite（异步）
    await SQLiteAdapter.setSetting(key, value);
  }

  // 内存缓存管理（LRU）
  setMemoryCache(key, value) {
    if (this.memoryCache.size >= this.maxMemoryCacheSize) {
      // 删除最旧的
      const firstKey = this.memoryCache.keys().next().value;
      this.memoryCache.delete(firstKey);
    }
    this.memoryCache.set(key, value);
  }

  // 清空缓存
  clear() {
    this.memoryCache.clear();
  }
}

export default new CacheManager();
```

```javascript
// src/services/storage/MigrationService.js
class MigrationService {
  async migrateFromAsyncStorageToSQLite() {
    try {
      console.log('开始数据迁移...');

      // 1. 读取AsyncStorage中的所有图片数据
      const imagesJson = await AsyncStorage.getItem('images');
      if (!imagesJson) {
        console.log('没有需要迁移的数据');
        return;
      }

      const images = JSON.parse(imagesJson);
      console.log(`发现 ${images.length} 条数据需要迁移`);

      // 2. 批量写入SQLite
      await SQLiteAdapter.batchInsertImages(images);

      // 3. 迁移设置
      const settings = await AsyncStorage.getItem('settings');
      if (settings) {
        const settingsObj = JSON.parse(settings);
        for (const [key, value] of Object.entries(settingsObj)) {
          await SQLiteAdapter.setSetting(key, JSON.stringify(value));
        }
      }

      // 4. 备份AsyncStorage数据（可选）
      await AsyncStorage.setItem('images_backup', imagesJson);
      await AsyncStorage.setItem('migration_date', new Date().toISOString());

      // 5. 清空AsyncStorage（可选，建议保留一段时间）
      // await AsyncStorage.removeItem('images');

      console.log('数据迁移完成');
    } catch (error) {
      console.error('数据迁移失败:', error);
      throw error;
    }
  }

  // 检查是否需要迁移
  async needsMigration() {
    // 检查SQLite是否有数据
    const sqliteHasData = await SQLiteAdapter.hasData();
    if (sqliteHasData) return false;

    // 检查AsyncStorage是否有数据
    const asyncStorageData = await AsyncStorage.getItem('images');
    return !!asyncStorageData;
  }
}

export default new MigrationService();
```

| L1: 内存缓存 | Map | 热数据缓存 | 500条 |
| L2: SQLite | react-native-sqlite-storage | 持久化存储 | 无限制 |
| L3: 文件系统 | react-native-fs | 图片文件 | 手机存储 |
